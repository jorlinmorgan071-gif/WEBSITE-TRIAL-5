// ---------------------------------------------------------- // A) DOWNLOAD: Resolve a YouTube video URL to direct download streams // ---------------------------------------------------------- // Returns null if the video is blocked / restricted. // Throws a descriptive string if we can explain why. // ---------------------------------------------------------- Future<VideoDownloadInfo?> getDownloadInfo(String url) async { try { // 1. Parse video ID from any YouTube URL format final videoId = VideoId.parseVideoId(url); if (videoId == null) { throw const FormatException('INVALID_URL'); }

  // 2. Fetch video metadata
  final video = await _explode.videos.get(videoId.value);

  // 3. Check restrictions
  if (video.status?.isPremium ?? false) {
    throw const FormatException('PREMIUM_CONTENT');
  }
  if (video.status?.isRequiresPurchase ?? false) {
    throw const FormatException('PURCHASE_REQUIRED');
  }
  if (video.status?.isUnplayable ?? false) {
    throw const FormatException('BLOCKED_CONTENT');
  }

  // 4. Get stream manifest (contains all available qualities)
  final manifest = await _explode.videos.streamsClient.getManifest(videoId.value);

  // 5. Select best available streams
  //    - Prefer muxed (video+audio combined) if available
  //    - Fall back to separate best video + best audio
  StreamInfo? videoStream;
  StreamInfo? audioStream;

  // Try muxed first (simplest for playback)
  if (manifest.muxed.isNotEmpty) {
    // Pick the highest quality muxed stream that's ≤1080p
    videoStream = manifest.muxed
        .where((s) => s.videoQuality?.height != null && s.videoQuality!.height! <= 1080)
        .reduce((a, b) =>
            (a.videoQuality?.height ?? 0) >= (b.videoQuality?.height ?? 0) ? a : b);
  }

  // If no muxed, get separate video-only and audio-only streams
  if (videoStream == null && manifest.video.isNotEmpty) {
    videoStream = manifest.video
        .where((s) => s.videoQuality?.height != null && s.videoQuality!.height! <= 1080)
        .reduce((a, b) =>
            (a.videoQuality?.height ?? 0) >= (b.videoQuality?.height ?? 0) ? a : b);
  }
  if (manifest.audioOnly.isNotEmpty) {
    audioStream = manifest.audioOnly
        .reduce((a, b) => a.bitrate > b.bitrate ? a : b);
  }

  if (videoStream == null) {
    throw const FormatException('NO_AVAILABLE_STREAM');
  }

  // 6. Return download info
  return VideoDownloadInfo(
    videoId: videoId.value,
    title: video.title,
    author: video.author,
    duration: video.duration,
    thumbnailUrl: video.thumbnails.highResUrl,
    videoStreamUrl: videoStream.url.toString(),
    audioStreamUrl: audioStream?.url.toString(),
    suggestedFileName: _sanitizeFileName('${video.title}.mp4'),
    fileSize: videoStream.size.totalBytes,
    isMuxed: audioStream == null, // true if both audio+video combined
  );
} on FormatException catch (e) {
  // Re-throw our known errors with user-friendly messages
  rethrow;
} catch (e) {
  debugPrint('[YouTubeService] getDownloadInfo error: $e');
  return null;
}
}

// ---------------------------------------------------------- // B) ACTUAL DOWNLOAD: Stream bytes from YouTube to local file // ---------------------------------------------------------- Future<DownloadedVideo> downloadVideo( VideoDownloadInfo info, { required void Function(double progress) onProgress, }) async { // 1. Determine local save directory final dir = await getDownloadDirectory();

// 2. Create output file
final file = File('${dir.path}/${info.suggestedFileName}');

// 3. If muxed stream — single download
if (info.isMuxed && info.videoStreamUrl != null) {
  await _downloadStream(
    Uri.parse(info.videoStreamUrl!),
    file,
    info.fileSize,
    onProgress,
  );
} else {
  // 4. Separate video + audio: download both, then merge via FFmpeg
  final videoFile = File('${dir.path}/_temp_video_${info.videoId}.mp4');
  final audioFile = File('${dir.path}/_temp_audio_${info.videoId}.m4a');

  // Download video track
  await _downloadStream(
    Uri.parse(info.videoStreamUrl!),
    videoFile,
    info.fileSize,
    (p) => onProgress(p * 0.7), // video = 70% of progress
  );

  // Download audio track
  if (info.audioStreamUrl != null) {
    await _downloadStream(
      Uri.parse(info.audioStreamUrl!),
      audioFile,
      0, // unknown size for audio
      (p) => onProgress(0.7 + p * 0.3), // audio = remaining 30%
    );
  }

  // Merge using FFmpeg (bundled in the app — see setup notes)
  await _mergeVideoAudio(videoFile, audioFile, file, (p) {
    onProgress(0.7 + 0.3 * p);
  });

  // Clean up temp files
  if (await videoFile.exists()) await videoFile.delete();
  if (await audioFile.exists()) await audioFile.delete();
}

// 5. Return the downloaded video metadata
return DownloadedVideo(
  videoId: info.videoId,
  title: info.title,
  author: info.author,
  localPath: file.path,
  thumbnailUrl: info.thumbnailUrl,
  duration: info.duration,
  downloadedAt: DateTime.now(),
  fileSize: await file.length(),
);
}

// Helper: download bytes from a URL to a file with progress Future<void> _downloadStream( Uri url, File file, int totalBytes, void Function(double progress) onProgress, ) async { final client = HttpClient(); final request = await client.getUrl(url); final response = await request.close(); final sink = file.openWrite(); int bytesReceived = 0;

await for (final chunk in response) {
  sink.add(chunk);
  bytesReceived += chunk.length;
  if (totalBytes > 0) {
    onProgress(bytesReceived / totalBytes);
  }
}
await sink.close();
client.close();
}
